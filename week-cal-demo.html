/* ── 自然語言日期格式（大小寫不敏感）─────────────── */
WeekCal.init({
  startDate: 'last Monday',    // today 之前最近一個週一
  endDate:   'next Sunday',     // today 之後最近一個週日
  days: [
    { date: 'yesterday', sourceId: 'src-a' },
    { date: 'today',     sourceId: 'src-b' },
    { date: 'tomorrow',  sourceId: 'src-c' },
    { date: 'next Fri',  sourceId: 'src-d' },
    { date: 'last Tue',  disabled: true   },
  ],
});

/* ── 內建主題（6 個）───────────────────────────────── */
//  theme        today      selected   indicator
//  ──────────   ────────   ────────   ─────────
//  'dark'       warning    lavender   ocean      ← 預設
//  'light'      warning    indigo*    teal       ← 亮色底
//  'teal'       teal       indigo     sky
//  'ocean'      ocean      lavender   teal
//  'indigo'     indigo     sky        lavender
//  'warm'       orange     salmon     yellow

WeekCal.init({ theme: 'teal'   });
WeekCal.init({ theme: 'indigo' });  // 範例三使用此主題
WeekCal.init({ theme: 'warm'   });

// colors{} 在 theme 之上局部覆蓋（只改需要的色鍵）：
WeekCal.init({
  theme:  'teal',
  colors: { indicator: '#FF91D7' },   // pink dot
});

/* ── 自訂第七套主題 ─────────────────────────────────── */
WeekCal.themes.myBrand = {
  calBg:        '#0C0D0C',
  shell:        '#C6C7BD',
  dayNameColor: 'rgba(198,199,189,0.60)',
  todayDayName: '#B3DE73',  // special
  todayBg:      '#B3DE73',  // special
  todayText:    '#0C0D0C',
  selectedBg:   '#FF91D7',  // pink
  selectedText: '#0C0D0C',
  indicator:    '#EDA109',  // orange
  cellHover:    'rgba(179,222,115,0.10)',
  headerText:   '#C6C7BD',
};
WeekCal.init({ theme: 'myBrand' });

/* ── 讀取主題值 ─────────────────────────────────────── */
WeekCal.themes.teal.todayBg;     // '#0DA591'
WeekCal.themes.warm.indicator;  // '#E3D322'

/* ── 其他 API ──────────────────────────────────────── */
WeekCal.configure({ animation: 'slide' }); // 全域預設
WeekCal.select('#cal', '2026-09-19');       // 程式化選取
WeekCal.getConfig('#cal');                  // 取得設定
WeekCal.destroy('#cal');                   // 卸載，還原 divs
