// 引入方式
<script src="box-components.js"></script>

// 使用方式
<box-single
  caption="區塊標題"
  caption-pos="top"
  caption-align="left"
  caption-bg="bg"
  caption-color="lavender"
  bg-color="stone" bg-opacity="0.3"
  border="1px solid #95BDD7"
  padding="24" gap="16"
>
  <!-- 任何 HTML 或自訂元件 -->
</box-single>

// 全域配置（影響所有容器）
BoxConfig.set({
  gap: 20, radius: 8, capBg: 'lavender',
});
